{ config, lib, pkgs, modulesPath, ... }:

{
  imports = [
    <nixos-wsl/modules>
  ];

  wsl = {
    enable = true;
    defaultUser = "nixos";
    startMenuLaunchers = true;
    useWindowsDriver = true;
  };

  nix.settings.experimental-features = [
    "nix-command"
    "flakes"
  ];

  services.desktopManager.gnome.enable = true;
  services.displayManager.gdm.enable = false;

  programs.dconf.enable = true;
  hardware.graphics.enable = true;

  environment.systemPackages = with pkgs; [
    gnome-tweaks
    gnome-extension-manager
    dconf-editor
    pavucontrol
    mesa-demos
    vulkan-tools
    wl-clipboard

    (writeShellScriptBin "gnome-wsl" ''
      set -euo pipefail

      export XDG_SESSION_TYPE=wayland
      export XDG_CURRENT_DESKTOP=GNOME
      export XDG_SESSION_DESKTOP=gnome

      export GDK_BACKEND=wayland,x11
      export MOZ_ENABLE_WAYLAND=1

      export WAYLAND_DISPLAY="''${WAYLAND_DISPLAY:-wayland-0}"
      export DISPLAY="''${DISPLAY:-:0}"
      export PULSE_SERVER="''${PULSE_SERVER:-unix:/mnt/wslg/PulseServer}"

      runtime_dir="''${XDG_RUNTIME_DIR:-/run/user/$(id -u)}"
      export XDG_RUNTIME_DIR="$runtime_dir"

      echo "Startar GNOME på NixOS genom WSLg..."
      echo "Wayland: $XDG_RUNTIME_DIR/$WAYLAND_DISPLAY"

      if [ ! -S "$XDG_RUNTIME_DIR/$WAYLAND_DISPLAY" ]; then
        echo "WSLg:s Wayland-socket hittades inte."
        exit 1
      fi

      unset DBUS_SESSION_BUS_ADDRESS 2>/dev/null || true

      exec ${dbus}/bin/dbus-run-session -- \
        ${gnome-shell}/bin/gnome-shell \
          --devkit \
          --wayland \
          --no-x11
    '')
  ];

  system.stateVersion = "26.05";
}
