About OPENSUSE SLOWROLL - https://en.opensuse.org/Portal:Slowroll

openSUSE Slowroll uses a modified version of openSUSE Tumbleweed
to give a little more stability to users who don‘t want the 
bleeding edge but also don’t want to wait 12 months for a
minor release or 3 to 4 years for a major release like openSUSE
Leap. With big updates once per month, and continuous bug fixes
and security fixes as they come in, Slowroll finds a perfect
balance for those who simply don’t want to be forced to choose!
Currently, Slowroll is still in beta phase so please be mindful
of any bugs, glitches or any graphics that may tie in to
another version of openSUSE.  

About KDE 6 FOR OPENSUSE SLOWROLL - https://kde.org

#!/usr/bin/env bash

set -o pipefail

LOG_FILE="$HOME/.local/state/plasma-x410.log"
mkdir -p "$(dirname "$LOG_FILE")"

echo "==================================================" | tee "$LOG_FILE"
echo " KDE Plasma 6 genom X410 – openSUSE Slowroll"      | tee -a "$LOG_FILE"
echo "==================================================" | tee -a "$LOG_FILE"

if ! command -v startplasma-x11 >/dev/null 2>&1; then
    echo "Fel: startplasma-x11 saknas." | tee -a "$LOG_FILE"
    exit 1
fi

if ! command -v xdpyinfo >/dev/null 2>&1; then
    echo "Installera xdpyinfo:" | tee -a "$LOG_FILE"
    echo "sudo zypper install xdpyinfo" | tee -a "$LOG_FILE"
    exit 1
fi

GATEWAY_IP="$(ip route show default 2>/dev/null |
    awk '/^default/ {print $3; exit}')"

DNS_IP="$(awk '/^nameserver/ {print $2; exit}' /etc/resolv.conf)"

CANDIDATES=()

[ -n "${X410_DISPLAY:-}" ] && CANDIDATES+=("$X410_DISPLAY")
[ -n "$GATEWAY_IP" ] && CANDIDATES+=("${GATEWAY_IP}:0.0")
CANDIDATES+=("127.0.0.1:0.0")
[ -n "$DNS_IP" ] && CANDIDATES+=("${DNS_IP}:0.0")

SELECTED_DISPLAY=""

echo "Söker efter X410..." | tee -a "$LOG_FILE"

for TEST_DISPLAY in "${CANDIDATES[@]}"; do
    echo "Testar DISPLAY=$TEST_DISPLAY" | tee -a "$LOG_FILE"

    if timeout 3 xdpyinfo -display "$TEST_DISPLAY" \
        >/dev/null 2>&1; then
        SELECTED_DISPLAY="$TEST_DISPLAY"
        break
    fi
done

if [ -z "$SELECTED_DISPLAY" ]; then
    echo | tee -a "$LOG_FILE"
    echo "Kunde inte ansluta till X410." | tee -a "$LOG_FILE"
    echo "Kontrollera följande:" | tee -a "$LOG_FILE"
    echo "  1. X410 är startat i Desktop mode." | tee -a "$LOG_FILE"
    echo "  2. Public Access är aktiverat i X410." | tee -a "$LOG_FILE"
    echo "  3. X410 tillåts genom Windows-brandväggen." | tee -a "$LOG_FILE"
    echo | tee -a "$LOG_FILE"
    echo "Du kan även ange adressen manuellt:" | tee -a "$LOG_FILE"
    echo "X410_DISPLAY=WINDOWS_IP:0.0 plasma-x410" | tee -a "$LOG_FILE"
    exit 1
fi

export DISPLAY="$SELECTED_DISPLAY"

unset WAYLAND_DISPLAY
unset WAYLAND_SOCKET
unset DBUS_SESSION_BUS_ADDRESS

export XDG_SESSION_TYPE=x11
export XDG_CURRENT_DESKTOP=KDE
export XDG_SESSION_DESKTOP=plasma
export DESKTOP_SESSION=plasma
export KDE_FULL_SESSION=true

export QT_QPA_PLATFORM=xcb
export GDK_BACKEND=x11
export SDL_VIDEODRIVER=x11
export QT_X11_NO_MITSHM=1

# Stabilare rendering genom X410/WSL2
export LIBGL_ALWAYS_SOFTWARE=1

# Ljud genom WSLg
if [ -S /mnt/wslg/PulseServer ]; then
    export PULSE_SERVER=unix:/mnt/wslg/PulseServer
elif [ -S /mnt/wslg/runtime-dir/pulse/native ]; then
    export PULSE_SERVER=unix:/mnt/wslg/runtime-dir/pulse/native
fi

export XDG_RUNTIME_DIR="/run/user/$(id -u)"

if [ ! -d "$XDG_RUNTIME_DIR" ]; then
    sudo mkdir -p "$XDG_RUNTIME_DIR"
    sudo chown "$(id -u):$(id -g)" "$XDG_RUNTIME_DIR"
    sudo chmod 700 "$XDG_RUNTIME_DIR"
fi

echo | tee -a "$LOG_FILE"
echo "DISPLAY=$DISPLAY" | tee -a "$LOG_FILE"
echo "XDG_RUNTIME_DIR=$XDG_RUNTIME_DIR" | tee -a "$LOG_FILE"
echo "PULSE_SERVER=${PULSE_SERVER:-inte hittad}" | tee -a "$LOG_FILE"
echo "Startar Plasma..." | tee -a "$LOG_FILE"
echo | tee -a "$LOG_FILE"

dbus-run-session -- startplasma-x11 2>&1 | tee -a "$LOG_FILE"

exit "${PIPESTATUS[0]}"
